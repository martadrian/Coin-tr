"use client"

import type React from "react"
import { useState } from "react"
import { motion } from "framer-motion"

const CoinFlipCasino: React.FC = () => {
  const [result, setResult] = useState<string>("Flip the coin!")
  const [isFlipping, setIsFlipping] = useState<boolean>(false)
  const [coinSide, setCoinSide] = useState<"heads" | "tails">("heads")

  const flipCoin = (choice: "heads" | "tails") => {
    if (isFlipping) return

    setIsFlipping(true)
    setResult("Flipping...")

    const randomFlip: "heads" | "tails" = Math.random() < 0.5 ? "heads" : "tails"

    setTimeout(() => {
      setCoinSide(randomFlip)
      setIsFlipping(false)
      if (choice === randomFlip) {
        setResult("You Won! 🎉")
      } else {
        setResult("You Lost! ❌")
      }
    }, 1000)
  }

  return (
    <div
      className="flex flex-col items-center justify-center min-h-screen bg-cover bg-center bg-no-repeat text-white"
      style={{ backgroundImage: "url('https://i.postimg.cc/sXBGmgyX/file-BAd-CT2a-Ef-UFqb-Ekqyxpcpp.webp')" }}
    >
      <h1 className="text-4xl font-bold mb-8">Coin Flip Casino</h1>
      <motion.div
        className="w-40 h-40 mb-8"
        animate={{ rotateY: isFlipping ? 720 : 0 }}
        transition={{ duration: 1, ease: "easeInOut" }}
      >
        <img
          src={
            coinSide === "heads"
              ? "https://i.postimg.cc/WbmFXKRM/file-GVx-Ns5e-FBAj-Te4-Rtm5-PKb-W-1.webp"
              : "https://i.postimg.cc/Jn0X0vFG/tails.webp"
          }
          alt="Coin"
          className="w-full h-full object-contain"
        />
      </motion.div>
      <div className="space-x-4">
        <button
          onClick={() => flipCoin("heads")}
          className="px-6 py-2 text-lg bg-yellow-500 text-black rounded hover:bg-yellow-600 transition-colors"
          disabled={isFlipping}
        >
          Heads
        </button>
        <button
          onClick={() => flipCoin("tails")}
          className="px-6 py-2 text-lg bg-yellow-500 text-black rounded hover:bg-yellow-600 transition-colors"
          disabled={isFlipping}
        >
          Tails
        </button>
      </div>
      <p
        className={`text-2xl mt-8 ${result.includes("Won") ? "text-green-500" : result.includes("Lost") ? "text-red-500" : "text-white"}`}
      >
        {result}
      </p>
    </div>
  )
}

export default CoinFlipCasino
